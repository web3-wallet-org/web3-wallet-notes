# 第12章 去中心化应用（DApp）- Web3开发者精简版

> **核心定位**: DApp是运行在区块链上的应用，通过智能合约实现后端逻辑，前端通过Web3库与链上交互。

---

## 一、DApp 架构解析

### 传统 Web2 vs DApp 架构对比

```
┌─────────────────────────────────────────────────────────────┐
│                    Web2 应用架构                              │
│                                                              │
│  用户浏览器 → 前端(React) → 后端API(Node.js) → 数据库(PostgreSQL)
│                                      ↓                       │
│                                 中心化服务器                   │
│                                 - 可被关停                    │
│                                 - 数据可篡改                  │
│                                 - 单点故障                    │
└─────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────┐
│                    DApp 架构                                  │
│                                                              │
│  用户浏览器 → 前端(React) → Web3.js/ethers.js → 智能合约(EVM)
│                                      ↓                       │
│                                 去中心化网络                   │
│                                 - 全球节点运行                 │
│                                 - 数据不可篡改                │
│                                 - 抗审查                      │
└─────────────────────────────────────────────────────────────┘
```

### DApp 核心组件

| 组件 | 作用 | 常用技术 | 优先级 |
|------|------|---------|--------|
| **前端** | 用户界面 | React/Vue/Next.js | ⭐⭐⭐⭐⭐ |
| **Web3库** | 与区块链交互 | ethers.js/web3.js | ⭐⭐⭐⭐⭐ |
| **钱包** | 身份认证+签名 | MetaMask/WalletConnect | ⭐⭐⭐⭐⭐ |
| **智能合约** | 业务逻辑 | Solidity/Vyper | ⭐⭐⭐⭐⭐ |
| **节点服务** | RPC接入 | Infura/Alchemy/自建节点 | ⭐⭐⭐⭐ |
| **索引服务** | 链上数据查询 | The Graph/SubQuery | ⭐⭐⭐⭐ |
| **存储** | 去中心化存储 | IPFS/Arweave | ⭐⭐⭐ |

### 2. 调用智能合约

```javascript
// 合约地址和 ABI
const CONTRACT_ADDRESS = '0x1234567890123456789012345678901234567890';
const CONTRACT_ABI = [
  'function balanceOf(address) view returns (uint256)',
  'function transfer(address to, uint256 amount) returns (bool)',
  'event Transfer(address indexed from, address indexed to, uint256 value)'
];

// 初始化合约实例
async function getContract(signer) {
  return new ethers.Contract(CONTRACT_ADDRESS, CONTRACT_ABI, signer);
}

// 读取数据（无需Gas，无需签名）
async function getBalance(userAddress) {
  const contract = await getContract(provider);
  const balance = await contract.balanceOf(userAddress);
  return ethers.formatEther(balance); // 转换为ETH单位
}

// 写入数据（需要Gas，需要签名）
async function transfer(toAddress, amount) {
  const signer = await provider.getSigner();
  const contract = await getContract(signer);
  
  // 发送交易
  const tx = await contract.transfer(
    toAddress,
    ethers.parseEther(amount) // 转换为wei
  );
  
  console.log('交易已发送:', tx.hash);
  
  // 等待确认
  const receipt = await tx.wait();
  console.log('交易已确认，区块:', receipt.blockNumber);
  
  return receipt;
}
```

### 2. 智能合约实现

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

// 代币合约
contract MyToken is ERC20, Ownable {
    constructor() ERC20("MyToken", "MTK") {
        _mint(msg.sender, 1000000 * 10 ** decimals());
    }
    
    function mint(address to, uint256 amount) public onlyOwner {
        _mint(to, amount);
    }
}

// 质押合约
contract Staking {
    IERC20 public token;
    uint256 public rewardRate = 100; // 每天每1000代币奖励100代币
    
    struct StakeInfo {
        uint256 amount;
        uint256 rewardDebt;
        uint256 lastUpdateTime;
    }
    
    mapping(address => StakeInfo) public stakes;
    
    constructor(address _token) {
        token = IERC20(_token);
    }
    
    function stake(uint256 amount) external {
        require(amount > 0, "Amount must be > 0");
        
        StakeInfo storage user = stakes[msg.sender];
        updateReward(msg.sender);
        
        user.amount += amount;
        token.transferFrom(msg.sender, address(this), amount);
    }
    
    function unstake(uint256 amount) external {
        require(amount > 0, "Amount must be > 0");
        
        StakeInfo storage user = stakes[msg.sender];
        require(user.amount >= amount, "Insufficient staked");
        
        updateReward(msg.sender);
        user.amount -= amount;
        token.transfer(msg.sender, amount);
    }
    
    function claimReward() external {
        updateReward(msg.sender);
        
        StakeInfo storage user = stakes[msg.sender];
        uint256 reward = pendingReward(msg.sender);
        
        require(reward > 0, "No reward to claim");
        user.rewardDebt = user.amount * rewardRate / 1e18;
        
        token.transfer(msg.sender, reward);
    }
    
    function updateReward(address user) internal {
        StakeInfo storage stake = stakes[user];
        uint256 timeElapsed = block.timestamp - stake.lastUpdateTime;
        uint256 reward = stake.amount * timeElapsed * rewardRate / (86400 * 1e18);
        stake.rewardDebt += reward;
        stake.lastUpdateTime = block.timestamp;
    }
    
    function pendingReward(address user) public view returns (uint256) {
        StakeInfo storage stake = stakes[user];
        uint256 timeElapsed = block.timestamp - stake.lastUpdateTime;
        return stake.amount * timeElapsed * rewardRate / (86400 * 1e18) - stake.rewardDebt;
    }
}
```

import { useState, useEffect } from 'react';

  const [signer, setSigner] = useState(null);
  const [tokenContract, setTokenContract] = useState(null);
  const [stakingContract, setStakingContract] = useState(null);
  
  // ABI定义
  const TOKEN_ABI = [/* ... */];
  const STAKING_ABI = [/* ... */];
  
  // 连接钱包
  async function connect() {
    if (!window.ethereum) {
      alert('请安装 MetaMask');
      return;
    }
    
    try {
      const provider = new ethers.BrowserProvider(window.ethereum);
      const signer = await provider.getSigner();
      const account = await signer.getAddress();
      
      setProvider(provider);
      setSigner(signer);
      setAccount(account);
      
      // 初始化合约
      setTokenContract(new ethers.Contract(TOKEN_ADDRESS, TOKEN_ABI, signer));

    } catch (error) {
      console.error('连接失败:', error);
    }
  }
  
  // 监听账户变化
  useEffect(() => {
    if (window.ethereum) {
      window.ethereum.on('accountsChanged', (accounts) => {
        setAccount(accounts[0]);
      });
    }
  }, []);
  
  return {
    account,
    provider,
    signer,
    tokenContract,
    stakingContract,
    connect
  };
}

// components/Staking.jsx
import { useState } from 'react';
import { useWeb3 } from '../hooks/useWeb3';
import { ethers } from 'ethers';

export function Staking() {
  const { account, tokenContract, stakingContract, connect } = useWeb3();
  const [stakeAmount, setStakeAmount] = useState('');
  const [stakedAmount, setStakedAmount] = useState('0');
  const [reward, setReward] = useState('0');
  
  // 加载用户数据
  async function loadUserData() {
    if (!stakingContract || !account) return;
    
    const stakeInfo = await stakingContract.stakes(account);
    setStakedAmount(ethers.formatEther(stakeInfo.amount));
    
    const pending = await stakingContract.pendingReward(account);
    setReward(ethers.formatEther(pending));
  }
  
  // 质押
  async function handleStake() {
    try {
      // 先授权
      await tokenContract.approve(
        STAKING_ADDRESS,
        ethers.parseEther(stakeAmount)
      );
      
      // 执行质押
      const tx = await stakingContract.stake(
        ethers.parseEther(stakeAmount)
      );
      await tx.wait();
      
      alert('质押成功！');
      loadUserData();
    } catch (error) {
      console.error('质押失败:', error);
    }
  }
  
  // 领取奖励
  async function handleClaim() {
    try {
      const tx = await stakingContract.claimReward();
      await tx.wait();
      
      alert('领取成功！');
      loadUserData();
    } catch (error) {
      console.error('领取失败:', error);
    }
  }
  
  if (!account) {
    return <button onClick={connect}>连接钱包</button>;
  }
  
  return (
    <div>
      <h2>质押面板</h2>
      <p>已质押: {stakedAmount} MTK</p>
      <p>待领取奖励: {reward} MTK</p>
      
      <input
        type="number"
        value={stakeAmount}
        onChange={(e) => setStakeAmount(e.target.value)}
        placeholder="输入质押数量"
      />
      <button onClick={handleStake}>质押</button>
      <button onClick={handleClaim}>领取奖励</button>
    </div>
  );
}

### 使用 The Graph 查询链上数据

```graphql
# subgraph.yaml
specVersion: 0.0.4
description: My DApp Subgraph
repository: https://github.com/your/repo

schema:
  file: ./schema.graphql

dataSources:
  - kind: ethereum/contract
    name: MyToken
    network: mainnet
    source:
      address: "0x1234567890123456789012345678901234567890"
      abi: ERC20
    mapping:
      kind: ethereum/events
      apiVersion: 0.0.6
      language: wasm/assemblyscript
      entities:
        - Transfer
      abis:
        - name: ERC20
          file: ./abis/ERC20.json
      eventHandlers:
        - event: Transfer(indexed address,indexed address,uint256)
          handler: handleTransfer

# schema.graphql
type Transfer @entity {
  id: ID!
  from: Bytes!
  to: Bytes!
  value: BigInt!
  timestamp: BigInt!
  blockNumber: BigInt!
}

type User @entity {
  id: ID!
  address: Bytes!
  balance: BigInt!
  transfers: [Transfer!]! @derivedFrom(field: "from")
}

# src/mapping.ts
import { Transfer } from '../generated/MyToken/MyToken';
import { User, Transfer as TransferEntity } from '../generated/schema';

export function handleTransfer(event: Transfer): void {
  let transfer = new TransferEntity(event.transaction.hash.toHex());
  transfer.from = event.params.from;
  transfer.to = event.params.to;
  transfer.value = event.params.value;
  transfer.timestamp = event.block.timestamp;
  transfer.blockNumber = event.block.number;
  transfer.save();
  
  // 更新用户余额
  let userFrom = User.load(event.params.from.toHex());
  if (!userFrom) {
    userFrom = new User(event.params.from.toHex());
    userFrom.address = event.params.from;
    userFrom.balance = BigInt.fromI32(0);
  }
  userFrom.balance = userFrom.balance.minus(event.params.value);
  userFrom.save();
}
```

## 七、自测题

### 题目1：读取合约数据和写入合约数据有什么区别？

**答案**：
| 维度 | 读取（Call） | 写入（Transaction） |
|------|-------------|-------------------|
| **是否修改状态** | 否 | 是 |
| **是否需要Gas** | 否 | 是 |
| **是否需要签名** | 否 | 是 |
| **返回值** | 直接返回 | 返回交易哈希，需等待确认 |
| **速度** | 即时 | 需要区块确认（~15秒） |
| **示例** | `balanceOf()` | `transfer()` |

### 题目2：如何处理用户拒绝交易的情况？

**答案**：
```javascript
try {
  const tx = await contract.someFunction();
  await tx.wait();
} catch (error) {
  if (error.code === 'ACTION_REJECTED') {
    // 用户拒绝签名
    console.log('用户取消了交易');
  } else if (error.code === 'INSUFFICIENT_FUNDS') {
    // 余额不足
    console.log('Gas费不足');
  } else {
    // 其他错误
    console.error('交易失败:', error);
  }
}
```

### 题目3：为什么需要使用 The Graph 等索引服务？

**答案**：
1. **性能**：直接查询链上数据非常慢（需要遍历所有区块）
2. **复杂度**：链上查询API功能有限，无法进行复杂过滤和排序
3. **成本**：频繁调用RPC节点费用高
4. **体验**：GraphQL提供灵活的查询接口，前端开发更高效
5. **实时性**：Subgraph自动监听事件并索引，提供实时更新

export function useMultiChain() {
  const [currentChain, setCurrentChain] = useState('ethereum');
  
  async function switchChain(chainName) {
    const chain = CHAINS[chainName];
    
    try {
      await window.ethereum.request({
        method: 'wallet_switchEthereumChain',
        params: [{ chainId: chain.chainId }]
      });
      setCurrentChain(chainName);
    } catch (error) {
      console.error('切换链失败:', error);
    }
  }
  
  function getContractInstance(signer, contractName) {
    const chain = CHAINS[currentChain];
    const address = chain[`${contractName}Address`];
    const abi = CONTRACT_ABIS[contractName];
    
    return new ethers.Contract(address, abi, signer);
  }
  
  return {
    currentChain,
    switchChain,
    getContractInstance,
    chains: CHAINS
  };
}
